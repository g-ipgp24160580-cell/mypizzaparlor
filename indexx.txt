      /* 3D Scene & Station Rendering */
      init3D() {
        this.container = document.getElementById('canvas-container');
        this.scene = new THREE.Scene();
        this.scene.background = new THREE.Color(0x1a0f07);
        this.scene.fog = new THREE.FogExp2(0x1a0f07, 0.03);

        this.camera.position.set(0, 1.6, 3); // First-person view height

        this.renderer = new THREE.WebGLRenderer({ antialias: true });
        this.renderer.setSize(window.innerWidth, window.innerHeight);
        this.renderer.shadowMap.enabled = true;
        this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;
        this.container.appendChild(this.renderer.domElement);

        // Lighting
        const ambientLight = new THREE.AmbientLight(0xfff5ea, 0.6);
        this.scene.add(ambientLight);

        const mainLight = new THREE.PointLight(0xffddaa, 1.2, 18);
        mainLight.position.set(0, 3.5, 0);
        mainLight.castShadow = true;
        this.scene.add(mainLight);

        // Brick Oven Glowing Light
        this.ovenLight = new THREE.PointLight(0xff5500, 2, 8);
        this.ovenLight.position.set(-3.5, 1.2, -3.5);

        this.buildStations();

      buildPizzeriaRoom() {
        // Checkered Floor
        const floorGeo = new THREE.PlaneGeometry(12, 12);
        const canvas = document.createElement('canvas');
        canvas.width = 256; canvas.height = 256;
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = '#2c1e14'; ctx.fillRect(0,0,256,256);
        ctx.fillStyle = '#dfc299'; ctx.fillRect(0,0,128,128); ctx.fillRect(128,128,128,128);
        const floorTex = new THREE.CanvasTexture(canvas);
        floorTex.wrapS = THREE.RepeatWrapping; floorTex.wrapT = THREE.RepeatWrapping;
        floorTex.repeat.set(6, 6);
        
        const floorMat = new THREE.MeshStandardMaterial({ map: floorTex, roughness: 0.4 });
        const floor = new THREE.Mesh(floorGeo, floorMat);
        floor.rotation.x = -Math.PI / 2;
        floor.receiveShadow = true;
        this.scene.add(floor);

        // Walls
        const wallMat = new THREE.MeshStandardMaterial({ color: 0x5c3a21, roughness: 0.8 });
        
        const backWall = new THREE.Mesh(new THREE.PlaneGeometry(12, 5), wallMat);
        backWall.position.set(0, 2.5, -6);
        this.scene.add(backWall);

        const leftWall = new THREE.Mesh(new THREE.PlaneGeometry(12, 5), wallMat);
        leftWall.position.set(-6, 2.5, 0);
        leftWall.rotation.y = Math.PI / 2;
        this.scene.add(leftWall);

        const rightWall = new THREE.Mesh(new THREE.PlaneGeometry(12, 5), wallMat);
        rightWall.position.set(6, 2.5, 0);
        rightWall.rotation.y = -Math.PI / 2;
        this.scene.add(rightWall);
      }

      /* Audio & Speech Synthesizer */
      initAudio() {
        this.audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        this.isMuted = false;
      }

      playTone(freq, duration, type='sine') {
        if (this.isMuted) return;
        try {
          const osc = this.audioCtx.createOscillator();
          const gain = this.audioCtx.createGain();
          osc.type = type;
          osc.frequency.value = freq;
          gain.gain.setValueAtTime(0.15, this.audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, this.audioCtx.currentTime + duration);
          osc.connect(gain);
          gain.connect(this.audioCtx.destination);
          osc.start();
          osc.stop(this.audioCtx.currentTime + duration);
        } catch(e) {}
      }

      playSizzleSound() {
        if (this.isMuted) return;
        this.playTone(180, 0.8, 'sawtooth');
      }

      playSuccessSound() {
        if (this.isMuted) return;
        this.playTone(523.25, 0.15); // C5
        setTimeout(() => this.playTone(659.25, 0.15), 150); // E5
        setTimeout(() => this.playTone(783.99, 0.3), 300); // G5
      }

      toggleMute() {
        this.isMuted = !this.isMuted;
        alert(this.isMuted ? "Audio Muted" : "Audio Unmuted");
      }

      speakText(text) {
        if ('speechSynthesis' in window) {
          window.speechSynthesis.cancel();
          const utterance = new SpeechSynthesisUtterance(text);
          utterance.lang = 'en-US';
          
          if (this.difficulty === 'beginner') utterance.rate = 0.8;
          else if (this.difficulty === 'intermediate') utterance.rate = 1.0;
          else if (this.difficulty === 'advanced') utterance.rate = 1.2;

          window.speechSynthesis.speak(utterance);
        }
      }

        this.moveVector = { x: 0, z: 0 };
        this.pitch = 0; this.yaw = 0;

        window.addEventListener('keydown', (e) => this.keys[e.key.toLowerCase()] = true);
        window.addEventListener('keyup', (e) => this.keys[e.key.toLowerCase()] = false);

        // Interaction Key
        window.addEventListener('keydown', (e) => {
          if (e.key.toLowerCase() === 'e') this.handleInteraction();
        });

        // Mouse Look
        let isDragging = false;
        document.addEventListener('mousedown', () => isDragging = true);
        document.addEventListener('mouseup', () => isDragging = false);
        document.addEventListener('mousemove', (e) => {
          if (isDragging || document.pointerLockElement === document.body) {
            this.yaw -= e.movementX * 0.003;
            this.pitch -= e.movementY * 0.003;
            this.pitch = Math.max(-Math.PI/3, Math.min(Math.PI/3, this.pitch));
          }
        });
      }

      setupTouchControls() {
        if ('ontouchstart' in window || navigator.maxTouchPoints > 0) {
          document.getElementById('joystick-zone').style.display = 'block';
          document.getElementById('touch-look-zone').style.display = 'block';
          document.getElementById('action-btn').style.display = 'flex';

          // Virtual Joystick Setup
          const stick = document.getElementById('joystick-stick');
          const base = document.getElementById('joystick-base');
          let touchId = null;

          base.addEventListener('touchstart', (e) => {
            touchId = e.changedTouches[0].identifier;
          });

          base.addEventListener('touchmove', (e) => {
            for (let t of e.changedTouches) {
              if (t.identifier === touchId) {
                const rect = base.getBoundingClientRect();
                const centerX = rect.left + rect.width / 2;
                const centerY = rect.top + rect.height / 2;
                let dx = t.clientX - centerX;
                let dy = t.clientY - centerY;
                const dist = Math.min(40, Math.hypot(dx, dy));
                const angle = Math.atan2(dy, dx);
                
                stick.style.transform = `translate(${Math.cos(angle)*dist}px, ${Math.sin(angle)*dist}px)`;
                this.moveVector.x = (Math.cos(angle) * dist) / 40;
                this.moveVector.z = (Math.sin(angle) * dist) / 40;
              }
            }
          });

          const resetJoystick = () => {
            stick.style.transform = `translate(0px, 0px)`;
            this.moveVector.x = 0; this.moveVector.z = 0;
          };
          base.addEventListener('touchend', resetJoystick);
          base.addEventListener('touchcancel', resetJoystick);

          // Touch Look Zone
          let lookTouchId = null, lastTouchX = 0, lastTouchY = 0;
          const lookZone = document.getElementById('touch-look-zone');
          lookZone.addEventListener('touchstart', (e) => {
            lookTouchId = e.changedTouches[0].identifier;
            lastTouchX = e.changedTouches[0].clientX;
            lastTouchY = e.changedTouches[0].clientY;
          });
          lookZone.addEventListener('touchmove', (e) => {
            for (let t of e.changedTouches) {
              if (t.identifier === lookTouchId) {
                const dx = t.clientX - lastTouchX;
                const dy = t.clientY - lastTouchY;
                this.yaw -= dx * 0.005;
                this.pitch -= dy * 0.005;
                this.pitch = Math.max(-Math.PI/3, Math.min(Math.PI/3, this.pitch));
                lastTouchX = t.clientX; lastTouchY = t.clientY;
              }
            }
          });

          // Action Button
          document.getElementById('action-btn').addEventListener('click', () => this.handleInteraction());
        }
      }

      /* Workflow & Mission Mechanics */
      startGame() {
        this.studentName = document.getElementById('student-name').value || "Student 1";
        this.studentClass = document.getElementById('student-class').value || "Year 4";
        this.difficulty = document.getElementById('difficulty-select').value;
        
        document.getElementById('start-modal').style.display = 'none';
        this.loadNewOrder();
      }

      loadNewOrder() {
        this.currentOrder = this.ordersPool[Math.floor(Math.random() * this.ordersPool.length)];
        this.currentStep = 1;
        this.preparedPizza = { dough: false, sauce: false, toppings: [] };
        this.audioReplayCount = 0;
        
        // Reset Pizza Visual Mesh
        this.doughMesh.scale.set(1, 1, 1);
        this.sauceMesh.visible = false;
        
        // Clear Topping Visuals
        for (let i = this.pizzaBaseGroup.children.length - 1; i >= 2; i--) {
          this.pizzaBaseGroup.remove(this.pizzaBaseGroup.children[i]);
        }

        this.updateMissionUI("1. Walk to Customer Table & Listen");
      }

      updateMissionUI(text) {
        document.getElementById('mission-step').innerText = text;
      }

      handleInteraction() {
        const p = this.camera.position;
        
        // Step 1: Customer Order
        if (this.currentStep === 1 && p.distanceTo(this.stations.customer.position) < 2.5) {
          document.getElementById('order-modal').style.display = 'flex';
          
          if (this.difficulty === 'advanced') {
            document.getElementById('order-speech-text').innerText = "🔊 [Audio Only - Listen Carefully!]";
          } else {
            document.getElementById('order-speech-text').innerText = `"${this.currentOrder.speech}"`;
          }
          this.playCustomerAudio();
        }
        // Step 2 & 3: Prep Counter
        else if ((this.currentStep === 2 || this.currentStep === 3) && p.distanceTo(this.stations.prep.position) < 2.5) {
          if (!this.preparedPizza.dough) {
            this.preparedPizza.dough = true;
            this.doughMesh.scale.set(1.3, 0.5, 1.3); // Knead/flatten dough animation
            this.playTone(200, 0.2, 'square');
            alert("Dough flattened and kneaded!");
          } else if (!this.preparedPizza.sauce) {
            this.preparedPizza.sauce = true;
            this.sauceMesh.visible = true;
            this.playTone(300, 0.2, 'sine');
            alert("Tomato sauce spread on pizza base!");
            this.currentStep = 3;
            this.updateMissionUI("3. Select Toppings at Prep Counter");
            document.getElementById('topping-palette').style.display = 'flex';
          }
        }
        // Step 4: Wood Oven
        else if (this.currentStep === 4 && p.distanceTo(this.stations.oven.position) < 2.5) {
          this.updateMissionUI("4. Baking Pizza in Wood Oven...");
          this.playSizzleSound();
          setTimeout(() => {
            this.playSuccessSound();
            alert("Pizza cooked to golden brown perfection!");
            this.currentStep = 5;
            this.updateMissionUI("5. Serve Customer & Take Listening Quiz");
          }, 2500);
        }
        // Step 5: Serve Customer
        else if (this.currentStep === 5 && p.distanceTo(this.stations.customer.position) < 2.5) {
          this.openQuizModal();
        }
      }

      playCustomerAudio() {
        this.audioReplayCount++;
        this.speakText(this.currentOrder.speech);
      }

      replayAudio() {
        if (this.currentOrder) {
          this.playCustomerAudio();
        }
      }

      closeOrderModal() {
        document.getElementById('order-modal').style.display = 'none';
        this.currentStep = 2;
        this.updateMissionUI("2. Knead Dough & Spread Sauce at Prep Counter");
      }

      addTopping(type) {
        if (!this.preparedPizza.toppings.includes(type)) {
          this.preparedPizza.toppings.push(type);
          
          // Visual Topping Mesh
          let topColor = 0xffffff;
          if (type === 'pepperoni') topColor = 0xc0392b;
          else if (type === 'mushrooms') topColor = 0xd5c8b8;
          else if (type === 'olives') topColor = 0x2c3e50;
          else if (type === 'peppers') topColor = 0x27ae60;
          else if (type === 'basil') topColor = 0x2ecc71;

          const topMesh = new THREE.Mesh(
            new THREE.CylinderGeometry(0.05, 0.05, 0.02, 8),
            new THREE.MeshStandardMaterial({ color: topColor })
          );
          topMesh.position.set((Math.random()-0.5)*0.4, 0.04, (Math.random()-0.5)*0.4);
          this.pizzaBaseGroup.add(topMesh);
          this.playTone(400, 0.1);
        }
      }

      finishToppings() {
        document.getElementById('topping-palette').style.display = 'none';
        this.currentStep = 4;
        this.updateMissionUI("4. Put Pizza into Wood-Fired Oven");
      }

      /* Quiz & Evaluation System */
      openQuizModal() {
        const quizModal = document.getElementById('quiz-modal');
        const qContainer = document.getElementById('quiz-options-container');
        const qText = document.getElementById('quiz-question-text');

        this.currentQuestionIdx = 0;
        this.quizCorrectCount = 0;

        this.renderQuizQuestion();
        quizModal.style.display = 'flex';
      }

      renderQuizQuestion() {
        const qData = this.currentOrder.questions[this.currentQuestionIdx];
        document.getElementById('quiz-question-text').innerText = `Question ${this.currentQuestionIdx + 1}: ${qData.q}`;
        
        const qContainer = document.getElementById('quiz-options-container');
        qContainer.innerHTML = '';

        qData.options.forEach((opt, idx) => {
          const btn = document.createElement('button');
          btn.className = 'quiz-option';
          btn.innerText = opt;
          btn.onclick = () => this.handleAnswer(idx);
          qContainer.appendChild(btn);
        });
      }

      handleAnswer(selectedIdx) {
        const qData = this.currentOrder.questions[this.currentQuestionIdx];
        if (selectedIdx === qData.answer) {
          this.quizCorrectCount++;
          this.playTone(600, 0.15);
        } else {
          this.playTone(150, 0.2, 'sawtooth');
        }

        this.currentQuestionIdx++;
        if (this.currentQuestionIdx < this.currentOrder.questions.length) {
          this.renderQuizQuestion();
        } else {
          document.getElementById('quiz-modal').style.display = 'none';
          this.finishOrder();
        }
      }

      finishOrder() {
        const totalQ = this.currentOrder.questions.length;
        const accuracy = Math.round((this.quizCorrectCount / totalQ) * 100);
        const points = accuracy * 2;
        
        this.score += points;
        document.getElementById('score-val').innerText = this.score;

        // Log Data
        const logEntry = {
          time: new Date().toLocaleTimeString(),
          name: this.studentName,
          class: this.studentClass,
          level: this.difficulty,
          score: points,
          accuracy: `${accuracy}%`,
          replays: this.audioReplayCount
        };
        this.gameLogs.push(logEntry);

        // UI Complete Modal
        document.getElementById('complete-score').innerText = points;
        document.getElementById('complete-stars').innerText = accuracy >= 80 ? '⭐⭐⭐' : accuracy >= 50 ? '⭐⭐' : '⭐';
        document.getElementById('complete-modal').style.display = 'flex';
        this.playSuccessSound();
      }

      nextOrder() {
        document.getElementById('complete-modal').style.display = 'none';
        this.loadNewOrder();
      }

      /* Teacher Dashboard & Data Export */
      openTeacherDashboard() {
        const tbody = document.getElementById('teacher-table-body');
        tbody.innerHTML = '';

        this.gameLogs.forEach(log => {
          const tr = document.createElement('tr');
          tr.innerHTML = `
            <td>${log.time}</td>
            <td>${log.name} (${log.class})</td>
            <td>${log.level}</td>
            <td>${log.score}</td>
            <td>${log.accuracy}</td>
            <td>${log.replays}</td>
          `;
          tbody.appendChild(tr);
        });

        document.getElementById('teacher-modal').style.display = 'flex';
      }

      exportCSV() {
        if (this.gameLogs.length === 0) {
          alert("No student data logs available yet!");
          return;
        }

        let csvContent = "data:text/csv;charset=utf-8,Timestamp,Student Name,Class,Difficulty,Score,Accuracy,Audio Replays\n";
        this.gameLogs.forEach(l => {
          csvContent += `"${l.time}","${l.name}","${l.class}","${l.level}",${l.score},"${l.accuracy}",${l.replays}\n`;
        });

        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a");
        link.setAttribute("href", encodedUri);
        link.setAttribute("download", `Pizzeria_Listening_Results_${this.studentName.replace(/\s+/g, '_')}.csv`);
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
      }

      syncToGoogleSheets() {
        const webhookUrl = document.getElementById('webhook-url').value;
        if (!webhookUrl) {
          alert("Please enter a valid Google Apps Script Webhook URL!");
          return;
        }

        fetch(webhookUrl, {
          method: 'POST',
          mode: 'no-cors',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(this.gameLogs)
        }).then(() => alert("Data synced to Google Sheets!"))
          .catch(err => alert("Error syncing data: " + err));
      }

      /* Render Loop */
      animate() {
        requestAnimationFrame(() => this.animate());

        // Player Movement Logic
        const speed = 0.08;
        let moveX = 0, moveZ = 0;

        if (this.keys['w'] || this.keys['arrowup']) moveZ -= speed;
        if (this.keys['s'] || this.keys['arrowdown']) moveZ += speed;
        if (this.keys['a'] || this.keys['arrowleft']) moveX -= speed;
        if (this.keys['d'] || this.keys['arrowright']) moveX += speed;

        // Apply Touch Vector
        moveX += this.moveVector.x * speed;
        moveZ += this.moveVector.z * speed;

        // Apply Rotation to Movement Vector
        const cosY = Math.cos(this.yaw);
        const sinY = Math.sin(this.yaw);

        this.camera.position.x += moveX * cosY + moveZ * sinY;
        this.camera.position.z += -moveX * sinY + moveZ * cosY;

        // Bounds Clamp
        this.camera.position.x = Math.max(-5, Math.min(5, this.camera.position.x));
        this.camera.position.z = Math.max(-5, Math.min(5, this.camera.position.z));

        // Camera Orientation
        const euler = new THREE.Euler(this.pitch, this.yaw, 0, 'YXZ');
        this.camera.quaternion.setFromEuler(euler);

        // Highlight Crosshair when facing stations
        const crosshair = document.getElementById('crosshair');
        let facingStation = false;
        Object.values(this.stations).forEach(st => {
          if (this.camera.position.distanceTo(st.position) < 2.5) facingStation = true;
        });
        if (facingStation) crosshair.classList.add('active');
        else crosshair.classList.remove('active');

        this.renderer.render(this.scene, this.camera);
      }

      onWindowResize() {
        this.camera.aspect = window.innerWidth / window.innerHeight;
        this.camera.updateProjectionMatrix();
        this.renderer.setSize(window.innerWidth, window.innerHeight);
      }
    }

    // Initialize Game Instance
    let game;
    window.addEventListener('DOMContentLoaded', () => {
      game = new PizzeriaGame();
    });
  </script>
